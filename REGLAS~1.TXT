rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ─────────────────────────────────────────────────────────────
    //  CRM-JORGE — Reglas por rol
    //
    //  Antes: cualquier usuario logueado podia escribir Y BORRAR todo.
    //  Ahora: el borrado queda restringido y la configuracion solo la
    //  puede tocar el administrador.
    //
    //  IMPORTANTE: si sumas un vendedor nuevo, hay que agregar su email
    //  en la lista de VENDEDORES de aca abajo, si no no va a poder cargar
    //  nada. Es la contra de este metodo, pero es lo que da la seguridad.
    // ─────────────────────────────────────────────────────────────

    function autenticado() {
      return request.auth != null
        && request.auth.token.firebase.sign_in_provider == 'password';
    }

    function esAdmin() {
      return autenticado()
        && request.auth.token.email == 'jorge.ledesmagd@gmail.com';
    }

    function esVendedor() {
      return autenticado()
        && request.auth.token.email in [
             'jledes.tf@gmail.com',
             'jorge_500_df@gmail.com',
             'pablodellacasa13@gmail.com'
           ];
    }

    // Puede cargar y modificar datos de trabajo. Cualquier otro usuario
    // autenticado (por ejemplo un Gerente que se cree despues) queda
    // automaticamente en solo lectura, que es lo correcto.
    function puedeEscribir() {
      return esAdmin() || esVendedor();
    }

    // Nombre del vendedor tal como se guarda en el campo "vend" de cada
    // contacto. Sirve para que un vendedor solo pueda borrar lo suyo.
    function miNombre() {
      return request.auth.token.email == 'jledes.tf@gmail.com' ? 'Jorge'
           : request.auth.token.email == 'jorge_500_df@gmail.com' ? 'Chamu'
           : request.auth.token.email == 'pablodellacasa13@gmail.com' ? 'Pablo'
           : '';
    }

    // ── CONFIGURACION ──
    // Mensajes, catalogos, links, umbrales. Solo el admin la modifica.
    // Esta es la proteccion del servidor contra lo que paso antes.
    match /config/{doc} {
      allow read:  if autenticado();
      allow write: if esAdmin();
    }

    // ── USUARIOS ── Solo el admin da de alta o modifica.
    match /usuarios/{doc} {
      allow read:  if autenticado();
      allow write: if esAdmin();
    }

    // ── CONTACTOS ──
    // Todos cargan y editan; borrar solo el admin, o el vendedor sus propios
    // contactos (necesario para poder cancelar un alta a medio hacer).
    match /contactos/{doc} {
      allow read:            if autenticado();
      allow create, update:  if puedeEscribir();
      allow delete:          if esAdmin()
                             || (esVendedor() && resource.data.vend == miNombre());
    }

    // ── VISITAS ── Son historial: se cargan pero no se borran.
    match /visitas/{doc} {
      allow read:            if autenticado();
      allow create, update:  if puedeEscribir();
      allow delete:          if esAdmin();
    }

    // ── COMODATOS (freezers) ──
    match /comodatos/{doc} {
      allow read:            if autenticado();
      allow create, update:  if puedeEscribir();
      allow delete:          if esAdmin();
    }

    // ── GIRA ── Se arma y desarma todo el tiempo: escritura libre.
    match /gira/{doc} {
      allow read:  if autenticado();
      allow write: if puedeEscribir();
    }

    // ── RECORRIDO (rastro GPS) ── Se agrega, no se edita ni se borra.
    match /recorrido/{doc} {
      allow read:             if autenticado();
      allow create:           if puedeEscribir();
      allow update, delete:   if esAdmin();
    }

    // ── LOG ── Es la auditoria: se escribe pero nadie la altera.
    match /log/{doc} {
      allow read:            if autenticado();
      allow create:          if autenticado();
      allow update, delete:  if esAdmin();
    }
  }
}
